			type = alliance
			id = event_target:CENTRAL_AMERICAN_FACTION_CREATOR
			value = 100
		}
		add_ai_strategy = {
			type = support
			id = event_target:CENTRAL_AMERICAN_FACTION_CREATOR
			value = 100
		}

            id = cea.12
            hours = 8
            random_hours = 24
        }
    }

        add_ideas = CEA_the_central_american_defense_pact_idea
    }
}

                }
            }

        }
    }
    else = {
        set_temp_variable = {
            HON_ceiling = -50
        }

            HON_ceiling = HON_control_over_nation_var
        }
