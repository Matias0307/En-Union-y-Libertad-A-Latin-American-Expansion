CHL_initiate_balance_of_power = {
    set_power_balance = {
        id = CHL_nacista_communist_bop
        left_side = CHL_communist_side # will keep the last set side if not specified
        right_side = CHL_nacista_side # will keep the last set side if not specified
        set_value = 0.25 # if specified will set the value of the power balance, this option will override the effect of set_default so the two options should not be used together. if neither set_default or set_value is specified the current value of the power balance will not change
    }
}

CHL_set_up_nacista_bop_push = {
	set_temp_variable = {
		CHL_nacista_bop_push_temp = 0
	}
	if = {
		limit = {
			check_variable = {
				CHL_nacista_power > 0
			}
		}
		if = {
			limit = {
				NOT = {
					has_idea = CHL_slight_nacista_push_1
				}
			}
			add_ideas = CHL_slight_nacista_push_1
		}
		add_to_temp_variable = {
			CHL_nacista_bop_push_temp = 0.3
		}
	}
	if = {
		limit = {
			check_variable = {
				CHL_nacista_power > 1
			}
		}
		if = {
			limit = {
				NOT = {
					has_idea = CHL_slight_nacista_push_2
				}
			}
			add_ideas = CHL_slight_nacista_push_2
		}
		add_to_temp_variable = {
			CHL_nacista_bop_push_temp = 0.5
		}
	}
	if = {
		limit = {
			check_variable = {
				CHL_nacista_power > 2
			}
		}
		if = {
			limit = {
				NOT = {
					has_idea = CHL_slight_nacista_push_3
				}
			}
			add_ideas = CHL_slight_nacista_push_3
		}
		add_to_temp_variable = {
			CHL_nacista_bop_push_temp = 0.7
		}
	}
	set_variable = {
		CHL_nacista_bop_push_var = CHL_nacista_bop_push_temp
	}
}

CHL_increase_nacista_power_category_one_level = {
	set_temp_variable = {
		CHL_nacista_power_temp = CHL_nacista_power
	}
	add_to_temp_variable = {
		CHL_nacista_power_temp = 1
	}
	clamp_temp_variable = {
		var = CHL_nacista_power_temp
		min = 0
		max = 3
	}

	custom_effect_tooltip = CHL_nacista_power_category_change_tt

	custom_effect_tooltip = CHL_skip_one_line_tt

	CHL_increase_5_nacista = yes

	if = {
		limit = {
			check_variable = {
				CHL_communists_power = 3
			}
			check_variable = {
				CHL_nacista_power = 3
			}
		}
		add_ideas = CHL_major_opposition_influence
	}

	hidden_effect = {
		add_to_variable = {
			CHL_nacista_power = 1
		}
		clamp_variable = {
			var = CHL_nacista_power
			min = 0
			max = 3
		}
		CHL_set_up_nacista_bop_push = yes
	}
}

CHL_improve_CORFO_industrial_concern = {
	if = {
		limit = {
			has_idea = CHL_CORFO_1
		}
		swap_ideas = {
			remove_idea = CHL_CORFO_1
			add_idea = CHL_CORFO_2
		}
	}
	else_if = {
		limit = {
			has_idea = CHL_CORFO_2
		}
		swap_ideas = {
			remove_idea = CHL_CORFO_2
			add_idea = CHL_CORFO_3
		}
	}
	else_if = {
		limit = {
			has_idea = CHL_CORFO_3
		}
		swap_ideas = {
			remove_idea = CHL_CORFO_3
			add_idea = CHL_CORFO_4
		}
	}
	else_if = {
		limit = {
			has_idea = CHL_CORFO_4
		}
		swap_ideas = {
			remove_idea = CHL_CORFO_4
			add_idea = CHL_CORFO_5
		}
	}
	else = {
		add_stability = 0.03
	}
}
