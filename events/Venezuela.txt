## Shall we intervene in Venezuela?
country_event = {
    id = venezuela.5
    title = venezuela.5.t 
    desc = venezuela.5.d 
    picture = GFX_report_event_generic_army

    is_triggered_only = yes 

    option = { ## intervene 
        name = venezuela.5.a 
        ai_chance = {
            base = 40
            modifier = {
                factor = 3
                has_government = democratic
            }
            modifier = {
                factor = 2
                VEN = { is_in_faction_with = GER }
            }
        }
        create_wargoal = {
            type = puppet_wargoal_focus
            target = VEN 
        }
        add_ai_strategy = {
            type = declare_war
            id = VEN 
            value = 50
        }
        add_war_support = 0.05
        if = {
            limit = { original_tag = USA }
            USA_house_small_opposition = yes
            USA_senate_small_opposition = yes
        }
    }

    option = { ## Issue a stern warning
        name = venezuela.5.b 
        ai_chance = {
            base = 10
            modifier = {
                factor = 10
                has_government = VEN 
            }
        }
        add_political_power = 25
        VEN = { 
            country_event = {
                id = venezuela.6
                days = 1
                hours = 6
            }
        }
    }

    option = { # let them go
        name = venezuela.5.c
        add_stability = 0.05
        add_war_support = -0.05
    }
}
