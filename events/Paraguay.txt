add_namespace = paraguay

# People protesting against the laissez-faire economics
country_event = {
	id = paraguay.1
	title = paraguay.1.t
	desc = paraguay.1.d
	picture = GFX_report_event_generic_riot

	is_triggered_only = yes

	# Placate the people
	option = {
		name = paraguay.1.a
		add_ideas = PAR_dissatisfied_population
	}
	# Send the military to deal with them
	option = {
		name = paraguay.1.b
		random_list = {
			35 = {
				add_manpower = -1500
				PAR_move_5_military = yes
				custom_effect_tooltip = PAR_people_will_back_down_tt
			}
			65 = {
				start_civil_war = {
					ideology = democratic
					size = 0.4
				}
				hidden_effect = {
					random_other_country = {
						limit = {
							has_government = democratic
							original_tag = PAR
						}
						retire_country_leader = yes
						promote_character = PAR_armed_civilians
					}
				}
			}
		}
	}
}

# Franco assumes power uwu <3
country_event = {
	id = paraguay.3
	title = paraguay.3.t
	desc = paraguay.3.d
	picture = GFX_PAR_febrista_revolution

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			retire_country_leader = yes
			set_politics = {
				ruling_party = communism
			}
			promote_character = PAR_rafael_franco
			set_power_balance = {
				id = PAR_military_leader_bop
				right_side = PAR_franco_side 
			}
			PAR_move_70_leader = yes
		}
	}

	option = {
		name = paraguay.3.a
		add_stability = -0.03
		add_political_power = -20
		set_party_name = {
			ideology = communism
			long_name = "Unión Nacional Revolucionaria"
			name = "UNR"
		}
		effect_tooltip = {
			retire_country_leader = yes
			promote_character = PAR_rafael_franco
			set_power_balance = {
				id = PAR_military_leader_bop
				right_side = PAR_franco_side 
			}
		}
	}
}
