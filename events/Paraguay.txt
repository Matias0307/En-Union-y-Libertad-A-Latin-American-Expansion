# People protesting against the laissez-faire economics
country_event = {
	id = paraguay.1
	title = paraguay.1.t
	desc = paraguay.1.d
	picture = GFX_report_event_generic_riot

	is_triggered_only = yes

	# Placate the people
	option = {
		name = paraguay.1.a
		add_ideas = PAR_dissatisfied_population
		unlock_decision_tooltip = PAR_appease_the_people
	}
	# Send the military to deal with them
	option = {
		name = paraguay.1.b
		random_list = {
			35 = {
				add_manpower = -1500
				PAR_move_5_military = yes
				custom_effect_tooltip = PAR_people_will_back_down_tt
			}
			65 = {
				if = {
					limit = {
						NOT = {
							has_template = "Milicia"
						}
					}
					hidden_effect = {
						division_template = { 
							name = "Milicia" 
							division_names_group = SPAN_INF_02 
							is_locked = yes 
							regiments = { 
								infantry = { x = 0 y = 0 }  
								infantry = { x = 0 y = 1 } 
								infantry = { x = 0 y = 2 } 
							}
						}
					}
				}
				start_civil_war = {
					ideology = democratic
					size = 0.4
				}
				hidden_effect = {
					every_controlled_state = {
						create_unit = {
							division = "name = \"Milicias Leales\" division_template = \"Milicia\" start_experience_factor = 0.02 start_equipment_factor = 0.9" 
							owner = ROOT
							count = 2
						}
					}
					random_other_country = {
						limit = {
							has_government = democratic
							has_civil_war = yes
							original_tag = PAR
						}
						retire_country_leader = yes
						promote_character = PAR_armed_civilians
						every_controlled_state = {
							create_unit = {
								division = "name = \"Milicias Anti-Gobierno\" division_template = \"Milicia\" start_experience_factor = 0.02 start_equipment_factor = 0.9" 
								owner = PREV
								count = 2
							}
						}
					}
				}
			}
		}
	}
}

# Purchasing Formosa - (✿o‿o)

	id = paraguay.5
	title = paraguay.5.t
	desc = paraguay.5.d
	picture = GFX_report_event_hitler_handshake

	is_triggered_only = yes

	#purchase Formosa
	option = {
		name = paraguay.5.a
		ai_chance = {
            base = 80
            modifier = {
                add = 60
                has_opinion = {
                    target = FROM
                    value > 69
                }
            }
            modifier = {
                add = 40 
                num_divisions > 24
            }
            modifier = {
                add = 20
                num_of_civilian_factories > 12
            }
        }
        FROM = {
            transfer_state = 980
            random_owned_controlled_state = {
                limit = { industrial_complex > 1 }
                remove_building = {
                    type = industrial_complex
                    level = 2
                }
            }
            country_event = { id = peru.16 hours = 3 }
        }
        add_offsite_building = { 
            type = industrial_complex 
            level = 2
        }
	}
	#don't purchase formosa
	option = {
		name = paraguay.5.b
		ai_chance = {
            base = 40 
            modifier = {
                add = 60 
                has_opinion = {
                    target = FROM
                    value < 0
                }
            }
            modifier = {
                add = 40
                has_war = yes 
            }
        }
        FROM = {
            country_event = { id = peru.17 hours = 6 }
            effect_tooltip = {
				random_list = {
					50 = {
						create_wargoal = {
							target = ROOT 
							type = take_state_focus
							generator = { 980 }
						}
					}
					50 = { }
				}
			}
        }
	}
}

				ruling_party = neutrality

				}
			}
		}
		else = { add_stability = -0.01 }
		retire_character = PAR_jose_felix_estigarribia
	}
